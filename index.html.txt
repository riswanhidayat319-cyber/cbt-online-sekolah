        function cbtApp() {
            return {
                loginTab: 'admin',
                currentUser: null,
                adminTab: 'rekap',
                showStudentModal: false,
                showQuestionModal: false,
                
                adminCredentials: {
                    username: 'Administrator',
                    password: 'password123'
                },
                adminForm: { username: '', password: '' },
                adminSettings: { username: 'Administrator', oldPassword: '', newPassword: '' },
                
                studentForm: { nis: '', token: '' },
                
                examConfig: {
                    class: '7A',
                    subject: 'Matematika Dasar',
                    token: 'CBT2026'
                },
                
                classList: ['7A', '7B', '8A', '8B', '9A', '9B'],
                
                students: [
                    { nis: '1001', name: 'Ahmad Fauzi', class: '7A' },
                    { nis: '1002', name: 'Siti Rahma', class: '7A' },
                    { nis: '1003', name: 'Budi Santoso', class: '7B' }
                ],
                
                questions: [
                    {
                        subject: 'Matematika Dasar',
                        question: 'Berapakah hasil dari 15 + 5 x 2?',
                        optA: '40',
                        optB: '25',
                        optC: '30',
                        optD: '20',
                        correct: 'B'
                    }
                ],
                
                newStudent: { nis: '', name: '', class: '7A' },
                newQuestion: { subject: '', question: '', optA: '', optB: '', optC: '', optD: '', correct: 'A' },
                
                studentAnswers: {},
                examResults: [],

                init() {
                    // Load from localStorage if available
                    if(localStorage.getItem('cbt_students')) this.students = JSON.parse(localStorage.getItem('cbt_students'));
                    if(localStorage.getItem('cbt_questions')) this.questions = JSON.parse(localStorage.getItem('cbt_questions'));
                    if(localStorage.getItem('cbt_results')) this.examResults = JSON.parse(localStorage.getItem('cbt_results'));
                    if(localStorage.getItem('cbt_admin')) this.adminCredentials = JSON.parse(localStorage.getItem('cbt_admin'));
                },

                adminLogin() {
                    if (this.adminForm.username === this.adminCredentials.username && this.adminForm.password === this.adminCredentials.password) {
                        this.currentUser = { role: 'Administrator', name: 'Administrator' };
                    } else {
                        alert('Username atau password admin salah!');
                    }
                },

                studentLogin() {
                    const found = this.students.find(s => s.nis === this.studentForm.nis);
                    if (!found) {
                        alert('NIS Siswa tidak ditemukan dalam Master Data!');
                        return;
                    }
                    if (this.studentForm.token.toUpperCase() !== this.examConfig.token.toUpperCase()) {
                        alert('Token ujian salah atau belum aktif!');
                        return;
                    }
                    this.currentUser = { role: 'Student', name: found.name, class: found.class, nis: found.nis };
                    this.studentAnswers = {};
                },

                logout() {
                    this.currentUser = null;
                    this.adminForm = { username: '', password: '' };
                    this.studentForm = { nis: '', token: '' };
                },

                openStudentModal() {
                    this.newStudent = { nis: '', name: '', class: '7A' };
                    this.showStudentModal = true;
                },

                saveStudent() {
                    this.students.push({...this.newStudent});
                    localStorage.setItem('cbt_students', JSON.stringify(this.students));
                    this.showStudentModal = false;
                    alert('Data siswa berhasil ditambahkan!');
                },

                deleteStudent(index) {
                    if(confirm('Hapus siswa ini?')) {
                        this.students.splice(index, 1);
                        localStorage.setItem('cbt_students', JSON.stringify(this.students));
                    }
                },

                openQuestionModal() {
                    this.newQuestion = { subject: this.examConfig.subject, question: '', optA: '', optB: '', optC: '', optD: '', correct: 'A' };
                    this.showQuestionModal = true;
                },

                saveQuestion() {
                    this.questions.push({...this.newQuestion});
                    localStorage.setItem('cbt_questions', JSON.stringify(this.questions));
                    this.showQuestionModal = false;
                    alert('Soal berhasil ditambahkan ke bank soal!');
                },

                deleteQuestion(index) {
                    if(confirm('Hapus soal ini?')) {
                        this.questions.splice(index, 1);
                        localStorage.setItem('cbt_questions', JSON.stringify(this.questions));
                    }
                },

                generateToken() {
                    const chars = 'ABCDEFGHJKLMNPQRSTUVWXYZ23456789';
                    let token = '';
                    for (let i = 0; i < 6; i++) {
                        token += chars.charAt(Math.floor(Math.random() * chars.length));
                    }
                    this.examConfig.token = token;
                },

                saveExamConfig() {
                    alert('Sesi ujian dan token berhasil diaktifkan: ' + this.examConfig.token);
                },

                updateAdminPassword() {
                    if(this.adminSettings.oldPassword !== this.adminCredentials.password) {
                        alert('Password lama salah!');
                        return;
                    }
                    this.adminCredentials.username = this.adminSettings.username;
                    this.adminCredentials.password = this.adminSettings.newPassword;
                    localStorage.setItem('cbt_admin', JSON.stringify(this.adminCredentials));
                    alert('Otoritas & password admin berhasil diperbarui!');
                    this.adminSettings.oldPassword = '';
                    this.adminSettings.newPassword = '';
                },

                submitExam() {
                    let correctCount = 0;
                    this.questions.forEach((q, idx) => {
                        if(this.studentAnswers[idx] && this.studentAnswers[idx] === q.correct) {
                            correctCount++;
                        }
                    });
                    const finalScore = this.questions.length > 0 ? Math.round((correctCount / this.questions.length) * 100) : 0;
                    
                    const resultEntry = {
                        name: this.currentUser.name,
                        class: this.currentUser.class,
                        subject: this.examConfig.subject,
                        score: finalScore,
                        timestamp: new Date().toLocaleString()
                    };

                    this.examResults.push(resultEntry);
                    localStorage.setItem('cbt_results', JSON.stringify(this.examResults));
                    alert('Ujian selesai! Nilai Anda: ' + finalScore);
                    this.logout();
                }
            }
        }
    </script>
</body>
</html>
